> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Trả lời 4 câu hỏi rủi ro của spec bằng phép thử thật (sinh ảnh, Postgres/pgmq/LangGraph, render Remotion trong Docker, upload YouTube) và ghi kết luận để cập nhật spec trước M1.

```js
import { mkdir, readFile, writeFile } from "node:fs/promises";

const BASE = process.env.NINE_ROUTER_BASE_URL;
const KEY = process.env.NINE_ROUTER_API_KEY;
const MODEL = process.env.IMAGE_MODEL ?? "ag/gemini-3.1-flash-image";
const OUT = new URL("../../../data/spikes/m0/", import.meta.url);

if (!BASE || !KEY) throw new Error("Thiếu NINE_ROUTER_BASE_URL hoặc NINE_ROUTER_API_KEY trong .env");

const CHARACTER =
  "A 15-year-old boy named Minh, 2D flat cartoon style, thick clean black outlines, " +
  "round face, short spiky black hair, orange hoodie, blue jeans, white sneakers, " +
  "full body, front view, standing, centered, no text";

async function generate(name, prompt, extra = {}) {
  const started = performance.now();
  const res = await fetch(`${BASE}/images/generations?response_format=binary`, {
    method: "POST",
    headers: { Authorization: `Bearer ${KEY}`, "Content-Type": "application/json" },
    body: JSON.stringify({ model: MODEL, prompt, size: "1024x1024", ...extra }),
  });
  if (!res.ok) throw new Error(`${name}: HTTP ${res.status} ${await res.text()}`);
  const type = res.headers.get("content-type") ?? "";
  const ext = type.includes("jpeg") ? "jpg" : "png";
  const buf = Buffer.from(await res.arrayBuffer());
  const file = new URL(`${name}.${ext}`, OUT);
  await writeFile(file, buf);
  console.log(`${name}: ${type}, ${buf.length} bytes, ${(performance.now() - started).toFixed(0)}ms`);
  return { file, type, buf };
}

await mkdir(OUT, { recursive: true });

await generate("a1-transparent", `${CHARACTER}, on a fully transparent background, PNG with alpha channel`);
const green = await generate(
  "a2-greenscreen",
  `${CHARACTER}, on a perfectly flat solid pure green (#00FF00) background, no shadows on the background`,
);
const ref = `data:${green.type || "image/png"};base64,${(await readFile(green.file)).toString("base64")}`;
await generate(
  "a3-happy-with-ref",
  `${CHARACTER}, big happy smile, on a perfectly flat solid pure green (#00FF00) background`,
  { image: ref },
);
await generate(
  "a4-happy-no-ref",
  `${CHARACTER}, big happy smile, on a perfectly flat solid pure green (#00FF00) background`,
);
```

- [ ] **Step 2: Viết `alpha-check.mjs`**

```js
import sharp from "sharp";

const [, , ...args] = process.argv;
const key = args.includes("--key");
const files = args.filter((a) => a !== "--key");
if (files.length === 0) throw new Error("Usage: alpha-check.mjs [--key] <image>...");

for (const file of files) {
  const meta = await sharp(file).metadata();
  const { data, info } = await sharp(file).ensureAlpha().raw().toBuffer({ resolveWithObject: true });
  const px = new Uint8ClampedArray(data.buffer, data.byteOffset, data.length);
  let transparent = 0;
  for (let i = 3; i < px.length; i += 4) if (px[i] < 16) transparent++;
  const total = info.width * info.height;
  console.log(
    `${file}: ${meta.format} ${info.width}x${info.height}, hasAlpha=${meta.hasAlpha}, ` +
      `transparent=${((transparent / total) * 100).toFixed(1)}%`,
  );

  if (!key) continue;
  let keyed = 0;
  for (let i = 0; i < px.length; i += 4) {
    const [r, g, b] = [px[i], px[i + 1], px[i + 2]];
    if (g > 140 && g > r * 1.4 && g > b * 1.4) {
      px[i + 3] = 0;
      keyed++;
    }
  }
  const out = file.replace(/\.(png|jpg)$/, "-keyed.png");
  await sharp(px, { raw: { width: info.width, height: info.height, channels: 4 } }).png().toFile(out);
  console.log(`  keyed ${((keyed / total) * 100).toFixed(1)}% -> ${out}`);
}
```

- [ ] **Step 2: Viết `db-probe.mjs`**

```js
import pg from "pg";
import { PostgresSaver } from "@langchain/langgraph-checkpoint-postgres";
import { Annotation, END, START, StateGraph } from "@langchain/langgraph";

const url = process.env.DATABASE_URL;
if (!url) throw new Error("Thiếu DATABASE_URL trong .env");
console.log("DATABASE_URL: đã đọc");

const QUEUE = "m0_probe";
const SCHEMA = "m0_probe_lg";
const pool = new pg.Pool({ connectionString: url, max: 3 });

async function q(label, sql, params = []) {
  const started = performance.now();
  const res = await pool.query(sql, params);
  console.log(`${label}: ${(performance.now() - started).toFixed(0)}ms, rows=${res.rowCount}`);
  return res;
}
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

try {
  await q("version", "select version()");

  // pgmq: tạo hàng đợi, kiểm tra thời gian ẩn (visibility timeout) để biết job lỗi có tự hiện lại không
  await q("create extension", "create extension if not exists pgmq");
  await q("create queue", "select pgmq.create($1)", [QUEUE]);
  const sent = await q("send", "select * from pgmq.send($1, $2::jsonb)", [QUEUE, JSON.stringify({ hello: "m0" })]);
  const msgId = sent.rows[0].send;

  const first = await q("read vt=2s", "select * from pgmq.read($1, $2, $3)", [QUEUE, 2, 1]);
  const hidden = await q("read ngay sau (mong đợi 0 dòng)", "select * from pgmq.read($1, $2, $3)", [QUEUE, 2, 1]);
  await sleep(3000);
  const again = await q("read sau 3s (mong đợi 1 dòng)", "select * from pgmq.read($1, $2, $3)", [QUEUE, 2, 1]);
  console.log(
    `pgmq: lần1=${first.rowCount}, ẩn=${hidden.rowCount}, hiện lại=${again.rowCount}, read_ct=${again.rows[0]?.read_ct}`,
  );
  await q("archive", "select pgmq.archive($1, $2::bigint)", [QUEUE, msgId]);

  // LangGraph checkpoint trong schema riêng để dọn sạch được
  await q("create schema", `create schema if not exists ${SCHEMA}`);
  const saver = new PostgresSaver(pool, undefined, { schema: SCHEMA });
  const setupStarted = performance.now();
  await saver.setup();
  console.log(`PostgresSaver.setup: ${(performance.now() - setupStarted).toFixed(0)}ms`);

  const State = Annotation.Root({ count: Annotation() });
  const graph = new StateGraph(State)
    .addNode("inc", (s) => ({ count: s.count + 1 }))
    .addEdge(START, "inc")
    .addEdge("inc", END)
    .compile({ checkpointer: saver });
  const config = { configurable: { thread_id: "m0" } };
  await graph.invoke({ count: 41 }, config);
  const snap = await graph.getState(config);
  console.log(`LangGraph state sau khi đọc lại từ Postgres: count=${snap.values.count} (mong đợi 42)`);
} finally {
  await pool.query("select pgmq.drop_queue($1)", [QUEUE]).catch((e) => console.log("drop queue:", e.message));
  await pool.query(`drop schema if exists ${SCHEMA} cascade`).catch((e) => console.log("drop schema:", e.message));
  console.log("Đã dọn hàng đợi và schema thử");
  await pool.end();
}
```

```tsx
import { AbsoluteFill, Img, interpolate, spring, staticFile, useCurrentFrame, useVideoConfig } from "remotion";

const LINES = [
  "Did you finish the homework?",
  "Homework? I thought that was optional.",
  "It was due yesterday.",
  "Then it's history homework now.",
];

export const Probe: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, width } = useVideoConfig();
  const beat = fps * 3;
  const x = width / 2 + Math.sin(frame / (fps * 2)) * 500;
  const bob = Math.abs(Math.sin(frame / 6)) * 20;
  const pop = spring({ fps, frame: frame % beat, config: { damping: 12 } });
  const fade = interpolate(frame % beat, [0, 8], [0, 1], { extrapolateRight: "clamp" });
  const line = LINES[Math.floor(frame / beat) % LINES.length];

  return (
```

- [ ] **Step 2: Viết `youtube-auth.mjs`**

```js
import http from "node:http";
import { mkdir, writeFile } from "node:fs/promises";
import { google } from "googleapis";

const id = process.env.YOUTUBE_OAUTH_CLIENT_ID;
const secret = process.env.YOUTUBE_OAUTH_CLIENT_SECRET;
if (!id || !secret) throw new Error("Thiếu YOUTUBE_OAUTH_CLIENT_ID / YOUTUBE_OAUTH_CLIENT_SECRET trong .env");

const PORT = 53682;
const REDIRECT = `http://127.0.0.1:${PORT}/callback`;
const OUT = new URL("../../../data/spikes/m0/", import.meta.url);
const TOKEN_FILE = new URL("youtube-token.json", OUT);

const oauth2 = new google.auth.OAuth2(id, secret, REDIRECT);
const url = oauth2.generateAuthUrl({
  access_type: "offline",
  prompt: "consent",
  scope: ["https://www.googleapis.com/auth/youtube.upload"],
});
console.log("Mở URL sau trong trình duyệt, đăng nhập đúng tài khoản kênh rồi cho phép:\n");
console.log(url);

const code = await new Promise((resolve, reject) => {
  const server = http
    .createServer((req, res) => {
      const u = new URL(req.url ?? "/", REDIRECT);
      if (u.pathname !== "/callback") {
        res.end();
        return;
      }
      const c = u.searchParams.get("code");
      res.end(c ? "Xong, quay lại terminal." : "Không nhận được code.");
      server.close();
      if (c) resolve(c);
      else reject(new Error(u.searchParams.get("error") ?? "no code"));
    })
    .listen(PORT, "127.0.0.1");
});

const { tokens } = await oauth2.getToken(code);
if (!tokens.refresh_token) throw new Error("Không nhận được refresh_token (thử thu hồi quyền app rồi chạy lại)");
await mkdir(OUT, { recursive: true });
await writeFile(TOKEN_FILE, JSON.stringify({ refresh_token: tokens.refresh_token, obtained_at: new Date().toISOString() }));
console.log(`Đã lưu refresh token vào ${TOKEN_FILE.pathname}`);
```
