```ts
// Chuẩn đặt tên asset (docs/standards/asset-naming.md). Sửa quy ước thì sửa cả hai nơi.

export const ASSET_KINDS = ["chr", "bg", "prop", "fx"] as const;
export type AssetKind = (typeof ASSET_KINDS)[number];

export const KIND_LABELS: Record<AssetKind, string> = {
  chr: "Nhân vật",
  bg: "Bối cảnh",
  prop: "Đạo cụ",
  fx: "Hiệu ứng",
};

export const STATES_BY_KIND: Record<AssetKind, readonly string[]> = {
  chr: ["base", "idle", "talk_a", "talk_b", "happy", "sad", "angry", "surprised", "walk"],
  bg: ["base", "day", "night", "rain"],
  prop: ["base", "broken"],
  fx: ["base"],
};

const NAME_RE = /^(chr|bg|prop|fx)_[a-z0-9]+(_[a-z0-9]+)*$/;
const MAX_NAME_LENGTH = 40;

export function isAssetKind(value: string): value is AssetKind {
  return (ASSET_KINDS as readonly string[]).includes(value);
}

// Trả về lý do sai, hoặc null nếu tên hợp lệ với loại asset.
export function assetNameError(name: string, kind: AssetKind): string | null {
  if (!NAME_RE.test(name)) return "Chỉ dùng chữ thường không dấu, số, `_`, bắt đầu bằng chr_/bg_/prop_/fx_";
  if (name.length > MAX_NAME_LENGTH) return `Tối đa ${MAX_NAME_LENGTH} ký tự`;
  if (!name.startsWith(`${kind}_`)) return `Tên phải bắt đầu bằng ${kind}_`;
  return null;
}

export function isAllowedState(kind: AssetKind, state: string): boolean {
  return STATES_BY_KIND[kind].includes(state);
}

// Biến chuỗi bất kỳ (LLM đề xuất) thành tên hợp lệ nhất có thể; vẫn phải kiểm tra lại bằng assetNameError.
export function toAssetName(raw: string, kind: AssetKind): string {
  const slug = raw
    .normalize("NFD")
    .replace(/[̀-ͯ]/g, "")
    .replace(/đ/gi, "d")
    .toLowerCase()
    .replace(/^(chr|bg|prop|fx)_/, "")
    .replace(/[^a-z0-9]+/g, "_")
    .replace(/^_+|_+$/g, "");
  return `${kind}_${slug || "asset"}`.slice(0, MAX_NAME_LENGTH).replace(/_+$/, "");
}
```

```ts
import sharp from "sharp";

// Xử lý ảnh sinh ra (kết luận M0, spec §5 bước 1):
// model không cho nền trong suốt, nên sinh trên nền #00FF00 rồi tách nền, khử ám xanh ở viền, cắt sát.

export type ProcessedImage = { data: Buffer; width: number; height: number };

// Ngưỡng đã thử ở M0: loại 82–84% điểm ảnh nền, giữ nguyên nhân vật.
function isKeyGreen(r: number, g: number, b: number): boolean {
  return g > 140 && g > r * 1.4 && g > b * 1.4;
}

export async function keyAndTrim(input: Buffer): Promise<ProcessedImage> {
  const { data, info } = await sharp(input).ensureAlpha().raw().toBuffer({ resolveWithObject: true });
  const { width, height } = info;
  const px = new Uint8ClampedArray(data.buffer, data.byteOffset, data.length);

  for (let i = 0; i < px.length; i += 4) {
    if (isKeyGreen(px[i], px[i + 1], px[i + 2])) px[i + 3] = 0;
  }

  // Khử ám xanh: điểm ảnh còn giữ nằm sát vùng đã tách thì kẹp kênh xanh về max(đỏ, lam).
  // Chỉ làm ở viền để không đổi màu áo/đồ vật màu xanh bên trong.
  const transparent = (x: number, y: number) =>
    x < 0 || y < 0 || x >= width || y >= height || px[(y * width + x) * 4 + 3] === 0;
  let minX = width;
  let minY = height;
  let maxX = -1;
  let maxY = -1;
  for (let y = 0; y < height; y++) {
    for (let x = 0; x < width; x++) {
      const i = (y * width + x) * 4;
      if (px[i + 3] === 0) continue;
      if (transparent(x - 1, y) || transparent(x + 1, y) || transparent(x, y - 1) || transparent(x, y + 1)) {
        px[i + 1] = Math.min(px[i + 1], Math.max(px[i], px[i + 2]));
      }
      if (x < minX) minX = x;
      if (x > maxX) maxX = x;
      if (y < minY) minY = y;
      if (y > maxY) maxY = y;
    }
  }
  if (maxX < 0) throw new Error("Ảnh rỗng sau khi tách nền (toàn bộ là nền xanh)");

  const trimmed = await sharp(Buffer.from(px.buffer, px.byteOffset, px.length), {
    raw: { width, height, channels: 4 },
  })
    .extract({ left: minX, top: minY, width: maxX - minX + 1, height: maxY - minY + 1 })
    .png()
    .toBuffer({ resolveWithObject: true });
  return { data: trimmed.data, width: trimmed.info.width, height: trimmed.info.height };
}

// Bối cảnh không tách nền: chỉ chuẩn hóa sang PNG.
export async function toPng(input: Buffer): Promise<ProcessedImage> {
  const out = await sharp(input).png().toBuffer({ resolveWithObject: true });
  return { data: out.data, width: out.info.width, height: out.info.height };
}
```

```ts
import type { ChatMessage } from "@/lib/ninerouter/client";

import type { AssetKind } from "./naming";

const STYLE = "2D flat cartoon style, thick clean black outlines, simple shading, no text, no watermark";
const GREEN_BG = "on a perfectly flat solid pure green (#00FF00) background, no shadow on the background";

const STATE_PHRASES: Record<string, string> = {
  base: "",
  idle: "standing relaxed, neutral expression",
  talk_a: "talking, mouth wide open",
  talk_b: "talking, mouth half open",
  happy: "big happy smile",
  sad: "sad expression, slumped shoulders",
  angry: "angry expression, frowning",
  surprised: "surprised expression, eyes wide open",
  walk: "walking pose, mid-step",
  day: "bright daytime lighting",
  night: "night time, dark blue lighting",
  rain: "rainy weather",
  broken: "broken, damaged",
};

// Prompt ảnh cho một biến thể. Mô tả asset giữ nguyên giữa các biến thể để nhân vật ít lệch nhất có thể.
export function imagePrompt(kind: AssetKind, description: string, state: string, feedback?: string): string {
  const parts = [description.trim()];
  const phrase = STATE_PHRASES[state];
  if (phrase) parts.push(phrase);
  const framing: Record<AssetKind, string> = {
    chr: "full body, front view, centered",
    bg: "wide 16:9 background scene, no characters",
    prop: "single object centered, nothing else in the frame",
    fx: "single effect centered, nothing else in the frame",
  };
  parts.push(framing[kind]);
  parts.push(STYLE);
  if (kind !== "bg") parts.push(GREEN_BG);
  if (feedback?.trim()) parts.push(`Fix these problems from the previous attempt: ${feedback.trim()}`);
  return parts.join(", ");
}

export type AssetProposal = { name: string; displayName: string; description: string };

// LLM viết lại mô tả và đề xuất tên. Kết quả phải được kiểm tra bằng parseProposal.
export function proposalMessages(kind: AssetKind, brief: string): ChatMessage[] {
  return [
    {
      role: "system",
      content:
        "You design reusable assets for a 2D cartoon series for English-speaking teens (14-18). " +
        "Return ONLY a JSON object with keys: name, displayName, description. " +
        `name: lowercase ASCII snake_case starting with "${kind}_", max 40 chars, no accents. ` +
        "displayName: short human-friendly name. " +
        "description: one detailed English paragraph (60-120 words) describing only fixed visual traits " +
        "(shape, colors, clothing, proportions), so the same asset can be redrawn consistently. " +
        "No background, pose or emotion in the description.",
    },
    { role: "user", content: `Asset kind: ${kind}\nBrief from the operator: ${brief}` },
  ];
}

export function parseProposal(raw: string): AssetProposal {
  const text = raw.trim().replace(/^```(?:json)?\s*/i, "").replace(/\s*```$/, "");
  const start = text.indexOf("{");
  const end = text.lastIndexOf("}");
  if (start < 0 || end < start) throw new Error("LLM không trả về JSON");
  const json = JSON.parse(text.slice(start, end + 1)) as Record<string, unknown>;
  const pick = (key: string) => (typeof json[key] === "string" ? (json[key] as string).trim() : "");
  const proposal = { name: pick("name"), displayName: pick("displayName"), description: pick("description") };
  if (!proposal.name || !proposal.displayName || !proposal.description) throw new Error("JSON thiếu trường bắt buộc");
  return proposal;
}
```

```ts
import { and, eq } from "drizzle-orm";

import { keyAndTrim, toPng } from "@/lib/assets/image";
import { assetNameError, isAllowedState, isAssetKind, toAssetName, type AssetKind } from "@/lib/assets/naming";
import { imagePrompt, parseProposal, proposalMessages } from "@/lib/assets/prompts";
import { getDb } from "@/lib/db/client";
import { assets, assetVariants } from "@/lib/db/schema";
import { chat, generateImage, requireModel } from "@/lib/ninerouter/client";
import type { JobPayload } from "@/lib/queue/jobs";
import { writeDataFile } from "@/lib/storage";

import { PermanentJobError } from "./job-types";

function str(payload: JobPayload, key: string): string {
  const value = payload[key];
  if (typeof value !== "string" || !value) throw new PermanentJobError(`Payload thiếu "${key}"`);
  return value;
}

// Tên đề xuất trùng thì thêm _2, _3...
async function uniqueName(base: string): Promise<string> {
  const db = getDb();
  for (let n = 1; n < 100; n++) {
    const candidate = n === 1 ? base : `${base.slice(0, 40 - `_${n}`.length)}_${n}`;
    const taken = await db.select({ id: assets.id }).from(assets).where(eq(assets.name, candidate)).limit(1);
    if (taken.length === 0) return candidate;
  }
  throw new Error(`Không tìm được tên trống cho ${base}`);
}

// Sinh ảnh cho một biến thể, xử lý, lưu file, chuyển sang chờ duyệt.
async function renderVariant(assetId: string, kind: AssetKind, description: string, state: string, feedback?: string) {
  const db = getDb();
  const prompt = imagePrompt(kind, description, state, feedback);
  const where = and(eq(assetVariants.assetId, assetId), eq(assetVariants.state, state));
  await db
    .insert(assetVariants)
    .values({ assetId, state, prompt, status: "generating" })
    .onConflictDoUpdate({
      target: [assetVariants.assetId, assetVariants.state],
      set: { prompt, status: "generating" },
    });

  try {
    const model = requireModel("IMAGE_MODEL");
    const image = await generateImage({ model, prompt });
    const processed = kind === "bg" ? await toPng(image.data) : await keyAndTrim(image.data);
    const filePath = `assets/${assetId}/${state}.png`;
    await writeDataFile(filePath, processed.data);
    await db
      .update(assetVariants)
      .set({ filePath, width: processed.width, height: processed.height, status: "pending_review" })
      .where(where);
    return { model, state, width: processed.width, height: processed.height, sourceMime: image.mime };
  } catch (error) {
    await db.update(assetVariants).set({ status: "failed" }).where(where);
    throw error;
  }
}

// Bước 1a: LLM viết mô tả và đề xuất tên, rồi sinh ảnh base.
// assetId do app tạo trước, nên chạy lại job không tạo asset trùng.
export async function generateBase(payload: JobPayload) {
  const assetId = str(payload, "assetId");
  const kind = str(payload, "kind");
  const brief = str(payload, "brief");
  if (!isAssetKind(kind)) throw new PermanentJobError(`Loại asset không hợp lệ: ${kind}`);
  const db = getDb();

  let [asset] = await db.select().from(assets).where(eq(assets.id, assetId)).limit(1);
  let usage: unknown = null;
  if (!asset) {
    const model = requireModel("LLM_WRITER_MODEL");
    const res = await chat({ model, messages: proposalMessages(kind, brief), temperature: 0.4 });
    usage = { model: res.model, ...res.usage };
    const proposal = parseProposal(res.content);
    let name = toAssetName(proposal.name, kind);
    if (assetNameError(name, kind)) name = toAssetName(brief, kind);
    name = await uniqueName(name);
    const displayName = typeof payload.displayName === "string" && payload.displayName ? payload.displayName : proposal.displayName;
    [asset] = await db
      .insert(assets)
      .values({ id: assetId, name, kind, displayName, description: proposal.description, status: "generating" })
      .returning();
  }

  const result = await renderVariant(asset.id, kind, asset.description, "base");
  await db.update(assets).set({ status: "pending_review" }).where(eq(assets.id, asset.id));
  return { name: asset.name, llm: usage, image: result };
}

// Sinh (lại) một biến thể. feedback = lý do bị loại lần trước.
export async function generateVariant(payload: JobPayload) {
  const assetId = str(payload, "assetId");
  const state = str(payload, "state");
  const feedback = typeof payload.feedback === "string" ? payload.feedback : undefined;
  const db = getDb();
  const [asset] = await db.select().from(assets).where(eq(assets.id, assetId)).limit(1);
  if (!asset) throw new PermanentJobError(`Không có asset ${assetId}`);
  if (!isAllowedState(asset.kind, state)) throw new PermanentJobError(`Trạng thái ${state} không hợp lệ cho ${asset.kind}`);
  const image = await renderVariant(asset.id, asset.kind, asset.description, state, feedback);
  if (state === "base") await db.update(assets).set({ status: "pending_review" }).where(eq(assets.id, asset.id));
  return { name: asset.name, image };
}
```

```ts
import { createReadStream } from "node:fs";
import path from "node:path";
import { Readable } from "node:stream";

import { dataFileInfo, dataPath } from "@/lib/storage";

export const dynamic = "force-dynamic";

const CONTENT_TYPES: Record<string, string> = {
  ".png": "image/png",
  ".jpg": "image/jpeg",
  ".webp": "image/webp",
  ".mp4": "video/mp4",
  ".json": "application/json",
};

// Trả file trong DATA_DIR cho trình duyệt (ảnh asset...). dataPath() chặn đường dẫn thoát ra ngoài.
export async function GET(_request: Request, { params }: { params: Promise<{ path: string[] }> }) {
  const relative = (await params).path.join("/");
  let full: string;
  try {
    full = dataPath(relative);
  } catch {
    return new Response("Not found", { status: 404 });
  }
  const info = await dataFileInfo(relative);
  if (!info?.isFile()) return new Response("Not found", { status: 404 });

  const body = Readable.toWeb(createReadStream(full)) as ReadableStream<Uint8Array>;
  return new Response(body, {
    headers: {
      "Content-Type": CONTENT_TYPES[path.extname(full).toLowerCase()] ?? "application/octet-stream",
      "Content-Length": String(info.size),
      "Cache-Control": "no-store",
    },
  });
}
```

```ts
import { and, asc, desc, eq, inArray } from "drizzle-orm";

import { getDb } from "@/lib/db/client";
import { assets, assetVariants, jobRuns } from "@/lib/db/schema";

export type Asset = typeof assets.$inferSelect;
export type AssetVariant = typeof assetVariants.$inferSelect;

export async function listAssets() {
  const db = getDb();
  const rows = await db.select().from(assets).orderBy(desc(assets.createdAt));
  const bases = rows.length
    ? await db
        .select()
        .from(assetVariants)
        .where(and(inArray(assetVariants.assetId, rows.map((r) => r.id)), eq(assetVariants.state, "base")))
    : [];
  const baseByAsset = new Map(bases.map((b) => [b.assetId, b]));
  return rows.map((asset) => ({ asset, base: baseByAsset.get(asset.id) ?? null }));
}

// Job tạo asset chưa có bản ghi assets (LLM chưa chạy xong) để danh sách hiện "đang tạo".
export async function listPendingAssetJobs() {
  return getDb()
    .select()
    .from(jobRuns)
    .where(and(eq(jobRuns.jobType, "asset.generate_base"), inArray(jobRuns.status, ["queued", "running"])))
    .orderBy(desc(jobRuns.createdAt));
}

export async function getAsset(id: string) {
  const db = getDb();
  const [asset] = await db.select().from(assets).where(eq(assets.id, id)).limit(1);
  if (!asset) return null;
  const variants = await db
    .select()
    .from(assetVariants)
    .where(eq(assetVariants.assetId, id))
    .orderBy(asc(assetVariants.createdAt));
  return { asset, variants };
}

// URL ảnh qua route /api/files; ?v= đổi khi ảnh được sinh lại để trình duyệt không dùng ảnh cũ.
export function variantImageUrl(variant: AssetVariant): string | null {
```

```ts
"use server";

import { randomUUID } from "node:crypto";
import { and, eq, ne } from "drizzle-orm";
import { revalidatePath } from "next/cache";
import { redirect } from "next/navigation";

import { assetNameError, isAllowedState, isAssetKind } from "@/lib/assets/naming";
import { getDb } from "@/lib/db/client";
import { assets, assetVariants, reviews } from "@/lib/db/schema";
import { enqueueJob } from "@/lib/queue/jobs";

export type FormState = { error?: string };

function field(form: FormData, key: string): string {
  const value = form.get(key);
  return typeof value === "string" ? value.trim() : "";
}

export async function createAsset(_prev: FormState, form: FormData): Promise<FormState> {
  const kind = field(form, "kind");
  const brief = field(form, "brief");
  const displayName = field(form, "displayName");
  if (!isAssetKind(kind)) return { error: "Chọn loại asset" };
  if (brief.length < 10) return { error: "Mô tả ngắn quá (ít nhất 10 ký tự)" };
  // assetId tạo trước để job chạy lại không tạo asset trùng.
  const assetId = randomUUID();
  await enqueueJob("asset.generate_base", { assetId, kind, brief, displayName }, { type: "asset", id: assetId });
  revalidatePath("/assets");
  redirect("/assets");
}

export async function approveVariant(_prev: FormState, form: FormData): Promise<FormState> {
  const variantId = field(form, "variantId");
  const db = getDb();
  const [variant] = await db.select().from(assetVariants).where(eq(assetVariants.id, variantId)).limit(1);
  if (!variant || variant.status !== "pending_review") return { error: "Biến thể không ở trạng thái chờ duyệt" };
  const [asset] = await db.select().from(assets).where(eq(assets.id, variant.assetId)).limit(1);
  if (!asset) return { error: "Không có asset này" };

  if (variant.state === "base") {
    // Duyệt ảnh gốc = chốt tên, tên hiển thị, mô tả. Tên không đổi được sau bước này.
    const name = field(form, "name");
    const displayName = field(form, "displayName");
    const description = field(form, "description");
    const nameError = assetNameError(name, asset.kind);
    if (nameError) return { error: `Tên không hợp lệ: ${nameError}` };
    if (!displayName || !description) return { error: "Thiếu tên hiển thị hoặc mô tả" };
    const clash = await db
      .select({ id: assets.id })
      .from(assets)
      .where(and(eq(assets.name, name), ne(assets.id, asset.id)))
      .limit(1);
    if (clash.length) return { error: `Tên ${name} đã có asset khác dùng` };
    await db.update(assets).set({ name, displayName, description, status: "approved" }).where(eq(assets.id, asset.id));
  }

  await db.update(assetVariants).set({ status: "approved" }).where(eq(assetVariants.id, variant.id));
  await db.insert(reviews).values({ targetType: "asset_variant", targetId: variant.id, decision: "approved" });
  revalidatePath(`/assets/${asset.id}`);
  revalidatePath("/assets");
  return {};
}

export async function rejectVariant(_prev: FormState, form: FormData): Promise<FormState> {
  const variantId = field(form, "variantId");
  const reason = field(form, "reason");
  if (!reason) return { error: "Ghi lý do loại để lần sinh sau sửa đúng chỗ" };
  const db = getDb();
  const [variant] = await db.select().from(assetVariants).where(eq(assetVariants.id, variantId)).limit(1);
  if (!variant || variant.status !== "pending_review") return { error: "Biến thể không ở trạng thái chờ duyệt" };

  await db.update(assetVariants).set({ status: "rejected" }).where(eq(assetVariants.id, variant.id));
  await db.insert(reviews).values({ targetType: "asset_variant", targetId: variant.id, decision: "rejected", reason });
  if (variant.state === "base") await db.update(assets).set({ status: "rejected" }).where(eq(assets.id, variant.assetId));
  // Sinh lại ngay, kèm lý do loại làm nhận xét cho prompt.
  await enqueueJob(
    "asset.generate_variant",
    { assetId: variant.assetId, state: variant.state, feedback: reason },
    { type: "asset_variant", id: variant.id },
  );
  revalidatePath(`/assets/${variant.assetId}`);
  return {};
}

export async function requestVariant(_prev: FormState, form: FormData): Promise<FormState> {
  const assetId = field(form, "assetId");
  const state = field(form, "state");
  const db = getDb();
  const [asset] = await db.select().from(assets).where(eq(assets.id, assetId)).limit(1);
  if (!asset) return { error: "Không có asset này" };
  if (asset.status !== "approved") return { error: "Duyệt ảnh gốc trước rồi mới sinh biến thể" };
  if (!isAllowedState(asset.kind, state) || state === "base") return { error: "Trạng thái không hợp lệ" };
  await enqueueJob("asset.generate_variant", { assetId, state }, { type: "asset", id: assetId });
  revalidatePath(`/assets/${assetId}`);
  return {};
}

// Sinh lại một biến thể bị lỗi (status failed), không cần lý do.
export async function retryVariant(_prev: FormState, form: FormData): Promise<FormState> {
  const assetId = field(form, "assetId");
  const state = field(form, "state");
  await enqueueJob("asset.generate_variant", { assetId, state }, { type: "asset", id: assetId });
  revalidatePath(`/assets/${assetId}`);
  return {};
}
```

```tsx
export function AutoRefresh({ active, intervalMs = 3000 }: { active: boolean; intervalMs?: number }) {
```

```tsx
        <CardHeader>
          <CardTitle>Mô tả</CardTitle>
          <CardDescription>Giữ nguyên cho mọi biến thể để nhân vật ít lệch nhất có thể.</CardDescription>
        </CardHeader>
        <CardContent className="text-sm whitespace-pre-wrap">{asset.description}</CardContent>
      </Card>

      <div className="grid gap-4 md:grid-cols-2 lg:grid-cols-3">
        {variants.map((variant) => {
          const url = variantImageUrl(variant);
          return (
            <Card key={variant.id}>
              <div className="flex h-64 items-center justify-center bg-[repeating-conic-gradient(#e5e5e5_0%_25%,transparent_0%_50%)] bg-[length:16px_16px]">
                {url && variant.status !== "generating" ? (
                  // eslint-disable-next-line @next/next/no-img-element -- ảnh động từ DATA_DIR, không qua tối ưu ảnh của Next
                  <img src={url} alt={`${asset.name}.${variant.state}`} className="max-h-60 max-w-full object-contain" />
                ) : (
                  <span className="text-xs text-muted-foreground">{STATUS_LABEL[variant.status]}</span>
                )}
              </div>
              <CardHeader>
                <CardTitle className="font-mono text-sm">
                  {asset.name}.{variant.state}
                </CardTitle>
                <CardDescription>
                  {STATUS_LABEL[variant.status]}
                  {variant.width && variant.height ? ` · ${variant.width}×${variant.height}` : ""}
                </CardDescription>
              </CardHeader>
              <CardContent>
                <VariantReview
                  assetId={asset.id}
                  variantId={variant.id}
                  state={variant.state}
                  status={variant.status}
                  base={
                    variant.state === "base" && asset.status !== "approved"
                      ? { name: asset.name, displayName: asset.displayName, description: asset.description }
                      : undefined
                  }
                />
              </CardContent>
            </Card>
          );
        })}
      </div>

      {asset.status === "approved" && (
        <Card>
          <CardHeader>
            <CardTitle>Thêm biến thể</CardTitle>
            <CardDescription>Trạng thái theo chuẩn đặt tên (docs/standards/asset-naming.md).</CardDescription>
          </CardHeader>
          <CardContent>
            <RequestVariantForm assetId={asset.id} states={missingStates} />
          </CardContent>
        </Card>
      )}
    </div>
  );
}
```
