## Hợp đồng dữ liệu (cần duyệt)

### Khuôn kịch bản `ScriptV1` (nằm trong `scripts.content.script`)

| Trường | Ràng buộc |
|---|---|
| `version` | literal `1` |
| `title` | 3–100 ký tự (tiêu đề tập, tiếng Anh) |
| `logline` | 10–300 ký tự |
| `targetSeconds` | int 30–600, bằng đầu vào của người dùng |
| `beats[]` | 3–30 nhịp |
| ↳ `id` | `b1`, `b2`… duy nhất |
| ↳ `role` | `hook` \| `setup` \| `escalation` \| `turn` \| `punchline` \| `resolution` \| `tag` |
| ↳ `summary` | 5–300 ký tự |
| ↳ `location` | `@bg_...[.state]` |
| ↳ `characters[]` | 1–4 × `@chr_...` (không ghi state) |
| ↳ `lines[]` | tối đa 12 × `{speaker: @chr_..., text 1–84 ký tự}`; `speaker` phải nằm trong `characters` của nhịp |
| ↳ `actions[]` | tối đa 8 ghi chú hành động (chuỗi 5–200 ký tự) |
| ↳ `gag` (tùy chọn) | `{kind: wordplay\|situational\|physical\|callback\|subversion\|absurd, setupBeat: id nhịp trước hoặc chính nhịp này, note 5–200}` |
| ↳ `seconds` | int 3–60 |

### Bọc ngoài `scripts.content` (jsonb)

```
{ meta: { assetNames: string[], targetSeconds: number, feedback?: string },
  script?: ScriptV1,      // có khi LLM trả JSON đúng khuôn
  issues?: string[] }     // lỗi lớp 1 của lần sinh cuối (nếu còn)
```

Khi `status = generating`, chỉ có `meta`. `meta` giữ đầu vào để sinh lại không cần hỏi lại người dùng.

### Khuôn `combos.definition` (combo cố định `default-v1`)

```
{ structure: [{role, minSeconds, maxSeconds, guidance}...],   // thứ tự nhịp
  hook: { maxSeconds: 15, guidance },                          // spec §7: hook 5–15 giây đầu
  gagTechniques: [{kind, guidance}...],
  tone: string, pacing: string, minGags: number }
```

Giá trị khởi điểm do code nạp (`ensureDefaultCombo`) nằm ở Task 1; sửa combo = sửa spec combo/ACE sau lát cắt đầu.

## Kiểm tra lớp 1 cho kịch bản (`checkScript`)

1. Đúng khuôn (zod). 2. `id` nhịp duy nhất; `gag.setupBeat` trỏ tới nhịp có thật, không đứng sau nhịp chứa gag. 3. Mọi `@...` là asset đã duyệt đúng loại; `speaker` thuộc `characters` của nhịp. 4. Nhịp đầu là `hook`, `seconds ≤ combo.hook.maxSeconds`. 5. Tổng `seconds` trong **±15%** `targetSeconds`. 6. Thoại của nhịp đọc kịp: `Σ max(5/6 s, ký tự/20) + 0.4 s mỗi câu ≤ seconds` của nhịp (dùng `estimateLineSeconds`). 7. Có ít nhất `combo.minGags` miếng hài. 8. Mỗi câu thoại ≤ 84 ký tự (đã nằm trong zod).

// (tra 2026-10-21). Khán giả 14-18 xếp vào nhóm người lớn (20 ký tự/giây). Đổi ngưỡng = đổi ở đây.
export const SUBTITLE = {
  maxCharsPerSecond: 20, // trẻ em: 17
  maxCharsPerLine: 42,
  maxLines: 2,
  minSeconds: 5 / 6,
  maxSeconds: 7,
} as const;

export const MAX_SUBTITLE_CHARS = SUBTITLE.maxCharsPerLine * SUBTITLE.maxLines; // 84

// Thời gian tối thiểu một câu cần hiện để đọc kịp (không nhỏ hơn mức tối thiểu của một phụ đề).
export function estimateLineSeconds(text: string): number {
  return Math.max(SUBTITLE.minSeconds, text.length / SUBTITLE.maxCharsPerSecond);
}

// Lấy một giá trị JSON từ phản hồi LLM (có thể bọc trong ```json ... ``` hoặc kèm lời dẫn).
export function extractJson(raw: string): unknown {
  const text = raw.trim().replace(/^```(?:json)?\s*/i, "").replace(/\s*```$/, "");
  const start = text.search(/[{[]/);
  const end = Math.max(text.lastIndexOf("}"), text.lastIndexOf("]"));
  if (start < 0 || end < start) throw new Error("LLM không trả về JSON");
  return JSON.parse(text.slice(start, end + 1));
}
```

import * as z from "zod";

import { MAX_SUBTITLE_CHARS } from "@/lib/validation/subtitle";

const NAME = "[a-z0-9]+(?:_[a-z0-9]+)*";
const chrRef = z.string().regex(new RegExp(`^@chr_${NAME}$`), "dạng @chr_ten");
const bgRef = z.string().regex(new RegExp(`^@bg_${NAME}(?:\\.${NAME})?$`), "dạng @bg_ten[.trang_thai]");

export const BEAT_ROLES = ["hook", "setup", "escalation", "turn", "punchline", "resolution", "tag"] as const;
export const GAG_KINDS = ["wordplay", "situational", "physical", "callback", "subversion", "absurd"] as const;

const beat = z.strictObject({
  id: z.string().regex(/^b\d{1,2}$/, "id dạng b1, b2…"),
  role: z.enum(BEAT_ROLES),
  summary: z.string().min(5).max(300),
  location: bgRef,
  characters: z.array(chrRef).min(1).max(4),
  lines: z.array(z.strictObject({ speaker: chrRef, text: z.string().min(1).max(MAX_SUBTITLE_CHARS) })).max(12),
  actions: z.array(z.string().min(5).max(200)).max(8),
  gag: z

  seconds: z.int().min(3).max(60),
});

import { eq } from "drizzle-orm";

import { getDb } from "@/lib/db/client";
import { combos } from "@/lib/db/schema";

import { comboDefinitionSchema, type ComboDefinition } from "./schema";

export const DEFAULT_COMBO_NAME = "default-v1";

// Combo cố định của lát cắt đầu (spec §7). Miếng hài theo "benign violation" (McGraw & Warren 2010):
// tình huống sai lệch nhưng vô hại. Sửa combo sau lát cắt đầu = spec combo/ACE.
export const DEFAULT_COMBO: ComboDefinition = {
  structure: [
    { role: "hook", minSeconds: 5, maxSeconds: 15, guidance: "Open on a striking question, problem or odd image that makes viewers want the answer." },
    { role: "setup", minSeconds: 10, maxSeconds: 40, guidance: "Establish the characters, the goal and one rule of the situation that will be bent later." },
    { role: "escalation", minSeconds: 15, maxSeconds: 60, guidance: "Each attempt makes things worse or weirder; plant the detail that pays off later." },
    { role: "punchline", minSeconds: 5, maxSeconds: 20, guidance: "Pay off the setup with a harmless violation of expectation." },
    { role: "resolution", minSeconds: 5, maxSeconds: 25, guidance: "Wrap up the situation in a clear final image." },
    { role: "tag", minSeconds: 3, maxSeconds: 10, guidance: "One last short joke or callback after the story ends." },
  ],
  hook: { maxSeconds: 15, guidance: "Within the first 5-15 seconds show the central conflict or question." },
  gagTechniques: [
    { kind: "callback", guidance: "Repeat an earlier detail in a new context." },
    { kind: "subversion", guidance: "Set up a familiar pattern, then break it." },
    { kind: "situational", guidance: "Ordinary people in a slightly absurd but harmless situation." },
    { kind: "wordplay", guidance: "Puns and double meanings that a teen audience gets instantly." },
  ],
  tone: "Witty, warm, a little absurd. Teen-friendly (TV-14 at most): no intense violence, sexual content, strong language or suggestive dialogue.",
  pacing: "Short punchy lines, a beat of action every few seconds, no scene longer than 25 seconds.",
  minGags: 2,
};

export async function ensureDefaultCombo(): Promise<{ id: string; definition: ComboDefinition }> {
  const db = getDb();
  const [existing] = await db.select().from(combos).where(eq(combos.name, DEFAULT_COMBO_NAME)).limit(1);
  if (existing) return { id: existing.id, definition: comboDefinitionSchema.parse(existing.definition) };
  const [created] = await db
    .insert(combos)
    .values({ name: DEFAULT_COMBO_NAME, definition: DEFAULT_COMBO })
    .onConflictDoNothing()
    .returning();
  if (created) return { id: created.id, definition: DEFAULT_COMBO };
  // Hai tiến trình tạo cùng lúc: đọc lại bản đã có.
  const [again] = await db.select().from(combos).where(eq(combos.name, DEFAULT_COMBO_NAME)).limit(1);
  return { id: again.id, definition: comboDefinitionSchema.parse(again.definition) };
}
```

```ts
import { isAllowedState } from "@/lib/assets/naming";
import type { Catalog } from "@/lib/scene/catalog";
import { parseRef } from "@/lib/scene/refs";
import { estimateLineSeconds } from "@/lib/validation/subtitle";

import type { ComboDefinition, ScriptV1 } from "./schema";

const TOLERANCE = 0.15; // tổng thời lượng lệch tối đa ±15% so với mục tiêu
const GAP_SECONDS = 0.4; // khoảng nghỉ giữa hai câu thoại

// Kiểm tra máy lớp 1 cho kịch bản (spec §6). Trả danh sách lỗi; rỗng = đạt.
export function checkScript(script: ScriptV1, catalog: Catalog, combo: ComboDefinition): string[] {
  const issues: string[] = [];
  const ids = new Set<string>();
  const index = new Map(script.beats.map((b, i) => [b.id, i]));

  const checkRef = (where: string, raw: string, kind: "chr" | "bg") => {
    const ref = parseRef(raw);
    const asset = ref && catalog.get(ref.name);
    if (!ref || ref.kind !== kind) return issues.push(`${where}: "${raw}" sai chuẩn tham chiếu`);
    if (!asset) return issues.push(`${where}: ${raw} không phải asset đã duyệt`);
    if (ref.state && !isAllowedState(ref.kind, ref.state)) issues.push(`${where}: trạng thái "${ref.state}" không hợp lệ`);
    else if (ref.state && !asset.variants.has(ref.state)) issues.push(`${where}: ${ref.name}.${ref.state} chưa có ảnh đã duyệt`);
  };

  script.beats.forEach((b, i) => {
    const where = `beats[${i}] (${b.id})`;
    if (ids.has(b.id)) issues.push(`${where}: id bị trùng`);
    ids.add(b.id);
    checkRef(`${where}.location`, b.location, "bg");
    b.characters.forEach((c, k) => checkRef(`${where}.characters[${k}]`, c, "chr"));
    b.lines.forEach((l, k) => {
      if (!b.characters.includes(l.speaker)) issues.push(`${where}.lines[${k}]: ${l.speaker} không có trong characters của nhịp`);
    });
    const needed = b.lines.reduce((sum, l) => sum + estimateLineSeconds(l.text) + GAP_SECONDS, 0);
    if (needed > b.seconds) {
      issues.push(`${where}: thoại cần ít nhất ${needed.toFixed(1)}s để đọc kịp nhưng nhịp chỉ có ${b.seconds}s`);
    }
    if (b.gag) {
      const setup = index.get(b.gag.setupBeat);
      if (setup === undefined) issues.push(`${where}.gag: setupBeat "${b.gag.setupBeat}" không tồn tại`);
      else if (setup > i) issues.push(`${where}.gag: setupBeat phải đứng trước hoặc là chính nhịp này`);
    }
  });

  const first = script.beats[0];
  if (first.role !== "hook") issues.push("Nhịp đầu tiên phải có role hook");
  else if (first.seconds > combo.hook.maxSeconds) issues.push(`Hook dài ${first.seconds}s, tối đa ${combo.hook.maxSeconds}s`);

  const total = script.beats.reduce((s, b) => s + b.seconds, 0);
  if (Math.abs(total - script.targetSeconds) > script.targetSeconds * TOLERANCE) {
    issues.push(`Tổng thời lượng ${total}s lệch quá ±15% so với mục tiêu ${script.targetSeconds}s`);
  }
  const gags = script.beats.filter((b) => b.gag).length;
  if (gags < combo.minGags) issues.push(`Cần ít nhất ${combo.minGags} miếng hài (gag), mới có ${gags}`);

  return issues;
}
```

import type { ChatMessage } from "@/lib/ninerouter/client";
import type { Catalog } from "@/lib/scene/catalog";

import { scriptSchema, type ComboDefinition } from "./schema";

export type ScriptPromptInput = {
  brief: string;
  targetSeconds: number;
  assetNames: string[];
  catalog: Catalog;
  combo: ComboDefinition;
  feedback?: string; // lý do bị loại (người) hoặc lỗi lớp 1 của lần trước

const clip = (text: string, n: number) => (text.length > n ? `${text.slice(0, n)}…` : text);

export function scriptMessages(input: ScriptPromptInput): ChatMessage[] {
  const assets = input.assetNames

import { eq } from "drizzle-orm";

import { getDb } from "@/lib/db/client";
import { scripts, validations, videos } from "@/lib/db/schema";
import { extractJson } from "@/lib/llm/json";
import { chat, requireModel } from "@/lib/ninerouter/client";
import type { JobPayload } from "@/lib/queue/jobs";
import { loadApprovedCatalog } from "@/lib/scene/catalog";
import { checkScript } from "@/lib/script/check";
import { ensureDefaultCombo } from "@/lib/script/combo";
import { scriptMessages } from "@/lib/script/prompts";
import { scriptContentSchema, scriptSchema, type ScriptContent } from "@/lib/script/schema";

import { PermanentJobError } from "./job-types";

const MAX_WRITES = 3; // lần viết đầu + tối đa 2 lần sinh lại khi sai lớp 1 (spec §6)
const RUBRIC = "l1-script-v1";

// Bước 2: viết kịch bản cho một phiên bản `scripts` đã được app tạo sẵn (idempotent theo scriptId).
export async function generateScript(payload: JobPayload) {
  const scriptId = typeof payload.scriptId === "string" ? payload.scriptId : "";
  if (!scriptId) throw new PermanentJobError('Payload thiếu "scriptId"');
  const db = getDb();
  const [row] = await db.select().from(scripts).where(eq(scripts.id, scriptId)).limit(1);
  if (!row) throw new PermanentJobError(`Không có script ${scriptId}`);
  const parsedContent = scriptContentSchema.safeParse(row.content);
  if (!parsedContent.success) throw new PermanentJobError("scripts.content thiếu meta");
  const { meta } = parsedContent.data;
  const [video] = await db.select().from(videos).where(eq(videos.id, row.videoId)).limit(1);
  if (!video) throw new PermanentJobError("Không có video của kịch bản này");

  const catalog = await loadApprovedCatalog();
  const missing = meta.assetNames.filter((n) => !catalog.has(n));
  if (missing.length) throw new PermanentJobError(`Asset chưa duyệt hoặc không tồn tại: ${missing.join(", ")}`);
  const { definition: combo } = await ensureDefaultCombo();
  const model = requireModel("LLM_WRITER_MODEL");

  await db.update(scripts).set({ status: "generating", model }).where(eq(scripts.id, scriptId));

  let feedback = meta.feedback;
  let previous: string | undefined;
  let usage: unknown[] = [];
  let finalContent: ScriptContent = { meta };
  let passed = false;

  for (let attempt = 1; attempt <= MAX_WRITES && !passed; attempt++) {
    const res = await chat({
      model,
      temperature: 0.8,
      timeoutMs: 180_000,
      messages: scriptMessages({
        brief: video.brief,
        targetSeconds: meta.targetSeconds,
        assetNames: meta.assetNames,
        catalog,
        combo,
        feedback,
        previous,
      }),
    });
    usage = [...usage, { attempt, model: res.model, ...res.usage }];

    let issues: string[];
    let script: ScriptContent["script"];
    try {
      const parsed = scriptSchema.safeParse(extractJson(res.content));
      if (parsed.success) {
        script = parsed.data;
        issues = checkScript(parsed.data, catalog, combo);
      } else {
        issues = parsed.error.issues.map((i) => `${i.path.join(".") || "(gốc)"}: ${i.message}`);
      }
    } catch (error) {
      issues = [`Phản hồi không phải JSON hợp lệ: ${error instanceof Error ? error.message : String(error)}`];
    }

    passed = issues.length === 0;
    await db.insert(validations).values({
      targetType: "script",
      targetId: scriptId,
      validatorModel: "layer1",
      rubricVersion: RUBRIC,
      scores: { attempt, issues },
      verdict: passed ? "pass" : "fail",
      feedback: issues.join("\n"),
    });
    finalContent = { meta, script, issues: passed ? undefined : issues };
    if (!passed) {
      previous = script ? JSON.stringify(script) : undefined;
      feedback = [meta.feedback, ...issues.slice(0, 15)].filter(Boolean).join("\n");
    }
  }

  await db
    .update(scripts)
    .set({ content: finalContent, status: passed ? "pending_review" : "failed" })
    .where(eq(scripts.id, scriptId));
  return { passed, attempts: usage.length, llm: usage };
}

export async function getVideo(id: string) {
  const db = getDb();
  const [video] = await db.select().from(videos).where(eq(videos.id, id)).limit(1);
  if (!video) return null;
  const versions = await db.select().from(scripts).where(eq(scripts.videoId, id)).orderBy(desc(scripts.version));

        .where(inArray(validations.targetId, versions.map((v) => v.id)))
        .orderBy(desc(validations.createdAt))
    : [];
  return { video, versions, checks };
}

async function startScriptVersion(videoId: string, meta: ScriptContent["meta"]) {

export async function createVideo(_prev: FormState, form: FormData): Promise<FormState> {
  const title = text(form, "title");
  const brief = text(form, "brief");
  const targetSeconds = Number(text(form, "targetSeconds"));
  const assetNames = form.getAll("assetNames").filter((v): v is string => typeof v === "string");
  if (title.length < 3) return { error: "Tiêu đề ít nhất 3 ký tự" };
  if (!Number.isInteger(targetSeconds) || targetSeconds < 30 || targetSeconds > 600) return { error: "Thời lượng 30–600 giây" };
  const catalog = await loadApprovedCatalog();
  const chosen = assetNames.filter((n) => catalog.has(n));
  if (!chosen.some((n) => catalog.get(n)?.kind === "chr")) return { error: "Chọn ít nhất 1 nhân vật đã duyệt" };
  if (!chosen.some((n) => catalog.get(n)?.kind === "bg")) return { error: "Chọn ít nhất 1 bối cảnh đã duyệt" };

  const combo = await ensureDefaultCombo();
  const [video] = await getDb().insert(videos).values({ title, brief, comboId: combo.id }).returning({ id: videos.id });
  await startScriptVersion(video.id, { assetNames: chosen, targetSeconds });
  revalidatePath("/videos");

async function loadScript(scriptId: string) {
  const [row] = await getDb().select().from(scripts).where(eq(scripts.id, scriptId)).limit(1);
  const content = row ? scriptContentSchema.safeParse(row.content) : null;
  return row && content?.success ? { row, content: content.data } : null;
}

export async function approveScript(_prev: FormState, form: FormData): Promise<FormState> {
  const found = await loadScript(text(form, "scriptId"));
  if (!found || found.row.status !== "pending_review") return { error: "Kịch bản không ở trạng thái chờ duyệt" };
  const db = getDb();
  await db.update(scripts).set({ status: "approved" }).where(eq(scripts.id, found.row.id));
  await db.update(videos).set({ status: "storyboarding" }).where(eq(videos.id, found.row.videoId));
  await db.insert(reviews).values({ targetType: "script", targetId: found.row.id, decision: "approved" });
  revalidatePath(`/videos/${found.row.videoId}`);
  return {};
}

export async function rejectScript(_prev: FormState, form: FormData): Promise<FormState> {
  const reason = text(form, "reason");
  if (reason.length < 5) return { error: "Ghi lý do loại (ít nhất 5 ký tự): lý do được đưa vào lần viết sau" };
  const found = await loadScript(text(form, "scriptId"));
  if (!found || !["pending_review", "failed"].includes(found.row.status)) return { error: "Kịch bản không thể loại ở trạng thái này" };
  const db = getDb();
  await db.update(scripts).set({ status: "rejected" }).where(eq(scripts.id, found.row.id));
  await db.insert(reviews).values({ targetType: "script", targetId: found.row.id, decision: "rejected", reason });
  await startScriptVersion(found.row.videoId, { ...found.content.meta, feedback: reason });
  revalidatePath(`/videos/${found.row.videoId}`);
  return {};
}

export async function regenerateScript(_prev: FormState, form: FormData): Promise<FormState> {

  if (!found || found.row.status !== "failed") return { error: "Chỉ sinh lại được kịch bản bị lỗi" };
  await startScriptVersion(found.row.videoId, { ...found.content.meta, feedback: undefined });
  revalidatePath(`/videos/${found.row.videoId}`);
  return {};
}
